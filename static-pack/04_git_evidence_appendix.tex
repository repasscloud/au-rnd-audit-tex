\section{Purpose}
\RDSectionIntro{This appendix is designed to import repository logs generated outside LaTeX so the claim pack can preserve a readable commit trail without manual copy and paste. Export the logs to plain-text files first, then compile this document.}

\RDAuditNote{A git log is not enough by itself. It supports chronology and technical traceability, but should be paired with notebook entries, ticket references, benchmark results, diagrams, and expenditure schedules.}

\section{Recommended export fields}
\begin{lstlisting}
%h | %ad | %an | %d | %s
\end{lstlisting}

\subsection{Example export command}
\begin{lstlisting}
git -C /path/to/repo log --since="2025-07-01" --until="2026-06-30" --date=short --pretty=format:"%h | %ad | %an | %d | %s" > evidence/git/repo-main.txt
\end{lstlisting}

\section{Commit interpretation notes}
\begin{RDChecklist}
\item Highlight commit clusters that align with discrete experiments.
\item Mark commits that were exploratory, benchmarking, instrumentation, or prototype related.
\item Link major merge ranges back to notebook run IDs or project IDs.
\item Exclude or separately note routine housekeeping, content-only changes, or unrelated operational work where appropriate.
\end{RDChecklist}
